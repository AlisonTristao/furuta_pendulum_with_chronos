cmake_minimum_required(VERSION 3.16.0)
# BTPDevice, BTP e TinyShell: submódulos git em extern/, fixados no commit testado com este firmware.
#   git clone --recursive ...   (ou, num clone já feito: git submodule update --init)
set(_extern ${CMAKE_CURRENT_LIST_DIR}/extern)
foreach(_dep BTPDevice BTP TinyShell)
    if(NOT EXISTS ${_extern}/${_dep}/CMakeLists.txt)
        message(FATAL_ERROR "extern/${_dep} vazio: rode  git submodule update --init  na raiz do repositório")
    endif()
    list(APPEND EXTRA_COMPONENT_DIRS ${_extern}/${_dep})
endforeach()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(furuta_esp32s3 VERSION 0.2.0)
